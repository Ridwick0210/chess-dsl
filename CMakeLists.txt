cmake_minimum_required(VERSION 3.16)
project(ChessScript LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(chessscript src/main.cpp src/lexer.cpp src/parser.cpp src/chess.cpp src/compiler.cpp src/report.cpp src/json.cpp)
target_include_directories(chessscript PRIVATE include)
if(MSVC)
  target_compile_options(chessscript PRIVATE /W4)
else()
  target_compile_options(chessscript PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(chess_rules tests/chess_rules.cpp src/chess.cpp src/lexer.cpp)
target_include_directories(chess_rules PRIVATE include)
if(MSVC)
  target_compile_options(chess_rules PRIVATE /W4)
else()
  target_compile_options(chess_rules PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME chess_rules COMMAND chess_rules)
if(UNIX)
  add_test(NAME compiler_smoke COMMAND sh tests/smoke.sh $<TARGET_FILE:chessscript>)
  set_tests_properties(compiler_smoke PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
